/* ==========================================
   EX.NO: 1(a)
   ARRAY IMPLEMENTATION OF STACK ADT
========================================== */
#include <stdio.h>
#include <conio.h>
#define max 5
static int stack[max];
int top = -1;

void push(int x) {
  if (top < max - 1) {
    stack[++top] = x;
  } else {
    printf("Stack overflow\n");
  }
}

int pop() {
  if (top >= 0) {
    return (stack[top--]);
  } else {
    printf("Stack underflow\n");
    return -1;
  }
}

void view() {
  int i;
  if (top < 0) {
    printf("Stack empty\n");
  } else {
    printf("Top-->");
    for (i = top; i >= 0; i--) {
      printf("%4d", stack[i]);
    }
    printf("\n");
  }
}

int main() {
  int ch = 0, val;
  while (ch != 4) {
    printf("1.Push 2.Pop 3.View 4.Quit\n");
    printf("Enter choice: ");
    scanf("%d", &ch);
    switch (ch) {
      case 1:
        printf("Enter element: ");
        scanf("%d", &val);
        push(val);
        break;
      case 2:
        val = pop();
        if (val != -1)
          printf("Popped element: %d\n", val);
        break;
      case 3:
        view();
        break;
      case 4:
        return 0;
      default:
        printf("Invalid choice\n");
    }
  }
  return 0;
}

/* ==========================================
   EX.NO: 2
   ARRAY IMPLEMENTATION OF QUEUE ADT
========================================== */
#include <stdio.h>
#define max 5
static int queue[max];
int front = -1, rear = -1;

void insert(int x) {
  if (rear < max - 1) {
    queue[++rear] = x;
    if (front == -1)
      front = 0;
  } else {
    printf("Queue full\n");
  }
}

int remove() {
  int val;
  if (front == -1) {
    printf("Queue empty\n");
    return -1;
  }
  val = queue[front];
  if (front == rear) {
    front = rear = -1;
  } else {
    front++;
  }
  return val;
}

void view() {
  int i;
  if (front == -1) {
    printf("Queue empty\n");
  } else {
    printf("Front-->");
    for (i = front; i <= rear; i++)
      printf("%4d", queue[i]);
    printf("<--rear\n");
  }
}

int main() {
  int ch = 0, val;
  while (ch != 4) {
    printf("1.Insert 2.Delete 3.View 4.Quit\n");
    printf("Enter choice: ");
    scanf("%d", &ch);
    switch (ch) {
      case 1:
        printf("Enter element: ");
        scanf("%d", &val);
        insert(val);
        break;
      case 2:
        val = remove();
        if (val != -1)
          printf("Deleted element: %d\n", val);
        break;
      case 3:
        view();
        break;
      case 4:
        return 0;
      default:
        printf("Invalid choice\n");
    }
  }
  return 0;
}

/* ==========================================
   EX.NO: 3
   ARRAY IMPLEMENTATION OF CIRCULAR QUEUE
========================================== */
#include <stdio.h>
#define maxsize 5
int a[maxsize];
int front = -1, rear = -1;

void enqueue(int x) {
  if ((front == 0 && rear == maxsize - 1) || (rear + 1) % maxsize == front) {
    printf("Queue is full\n");
  } else {
    if (front == -1) front = 0;
    rear = (rear + 1) % maxsize;
    a[rear] = x;
  }
}

void dequeue() {
  if (front == -1) {
    printf("Queue is empty\n");
  } else {
    if (front == rear) {
      front = rear = -1;
    } else {
      front = (front + 1) % maxsize;
    }
  }
}

void print() {
  int i, count;
  if (front == -1) {
    printf("Queue empty\n");
  } else {
    count = (rear + maxsize - front) % maxsize + 1;
    for (i = 0; i < count; i++) {
      printf("%d ", a[(front + i) % maxsize]);
    }
    printf("\n");
  }
}

int main() {
  enqueue(5);
  enqueue(7);
  enqueue(-3);
  enqueue(9);
  printf("After insertions:\n");
  print();
  dequeue();
  printf("After first deletion:\n");
  print();
  dequeue();
  printf("After second deletion:\n");
  print();
  enqueue(14);
  printf("After insertion:\n");
  print();
  return 0;
}
